async function loadDataFromSupabase() {

const opt = document.createElement('option');
opt.value = tourName;
opt.textContent = `${tourName} (${toursMap[tourName]})`;
select.appendChild(opt);
});
}
function processAndRenderStats(data) {
const citiesMap = {};
const venuesMap = {};
const countriesMap = {};
const yearsMap = {};
const categoriesMap = {};
const contributorsMap = {};
const songsMap = {};
const musiciansMap = {};
const toursMap = {};
let totalRecords = data.length;
let ticketCount = 0;
data.forEach(item => {
const cat = (item.KATEGORIE || 'tickets').trim();
const cleanCat = cat.charAt(0).toUpperCase() + cat.slice(1).toLowerCase();
categoriesMap[cleanCat] = (categoriesMap[cleanCat] || 0) + 1;
if (cleanCat.toLowerCase() === 'tickets') ticketCount++;
const datumText = item.DATUM || '';
const yearMatch = datumText.match(/\b(19\d\d|20\d\d)\b/);
if (yearMatch) {
const year = yearMatch[0];
yearsMap[year] = (yearsMap[year] || 0) + 1;
}
if (isValid(item.TOUR_NAME)) {
const tourName = item.TOUR_NAME.trim();
toursMap[tourName] = (toursMap[tourName] || 0) + 1;
}
if (isValid(item.MESTO)) {
const city = item.MESTO.trim();
citiesMap[city] = (citiesMap[city] || 0) + 1;
}
if (isValid(item.STAT)) {
const country = item.STAT.trim();
countriesMap[country] = (countriesMap[country] || 0) + 1;
}
if (isValid(item.VENUE)) {
const venue = item.VENUE.trim();
venuesMap[venue] = (venuesMap[venue] || 0) + 1;
}
if (isValid(item.CONTRIBUTOR)) {
const contrib = item.CONTRIBUTOR.trim();
contributorsMap[contrib] = (contributorsMap[contrib] || 0) + 1;
}
if (isValid(item.SETLIST)) {
const songs = item.SETLIST.split(',').map(s => s.trim()).filter(Boolean);
songs.forEach(song => {
if (song.length >= 2 && song.length < 60 && !song.startsWith('[')) {
songsMap[song] = (songsMap[song] || 0) + 1;
}
});
}
if (isValid(item.LINEUP)) {
const musicians = item.LINEUP.split(/[;/]/).map(m => m.trim()).filter(Boolean);
musicians.forEach(mus => {
const cleanName = mus.split('(')[0].trim();
if (cleanName.length > 3 && !cleanName.toLowerCase().includes('solo')) {
musiciansMap[cleanName] = (musiciansMap[cleanName] || 0) + 1;
}
});
}
});
document.getElementById('statRecords').textContent = totalRecords;
document.getElementById('statTickets').textContent = ticketCount;
document.getElementById('statCities').textContent = Object.keys(citiesMap).length;
document.getElementById('statVenues').textContent = Object.keys(venuesMap).length;
document.getElementById('statCountries').textContent = Object.keys(countriesMap).length;
document.getElementById('statContributors').textContent = Object.keys(contributorsMap).length;
populateTourDropdown(toursMap);
initMap();
updateMapTour();
renderYearBarChart(yearsMap);
renderBarChart('chartTours', toursMap, 10, '#38bdf8');
renderDoughnutChart('chartCategories', categoriesMap);
renderBarChart('chartContributors', contributorsMap, 10, '#c084fc');
renderBarChart('chartSongs', songsMap, 10, '#facc15');
renderBarChart('chartMusicians', musiciansMap, 10, '#4ade80');
renderBarChart('chartCities', citiesMap, 10, '#38bdf8');
renderBarChart('chartVenues', venuesMap, 10, '#f97316');
renderBarChart('chartCountries', countriesMap, 15, '#eab308');
}
function initMap() {
if (leafletMap) return;
leafletMap = L.map('worldMap').setView([30.0, 10.0], 2);
L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png', {
attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>',
maxZoom: 19
}).addTo(leafletMap);
mapMarkersGroup = L.layerGroup().addTo(leafletMap);
}
function updateMapTour() {
if (!leafletMap) return;
const selectedTour = document.getElementById('tourSelect')?.value || 'ALL';
mapMarkersGroup.clearLayers();
if (mapPolyline) {
leafletMap.removeLayer(mapPolyline);
mapPolyline = null;
}
const tourShows = globalRawData.filter(item => {
if (!isValid(item.MESTO)) return false;
const itemTour = (item.TOUR_NAME || '').trim();
if (selectedTour === 'ALL') return true;
if (selectedTour === 'NON_TOUR') return !itemTour;
return itemTour.toLowerCase() === selectedTour.toLowerCase();
}).sort((a, b) => (a.DATUM || '').localeCompare(b.DATUM || ''));
const routeCoords = [];
const bounds = [];
tourShows.forEach((show, idx) => {
const city = show.MESTO.trim();
let coords = CITY_COORDS[city];
if (!coords) {
const matchKey = Object.keys(CITY_COORDS).find(k => city.toLowerCase().includes(k.toLowerCase()) || k.toLowerCase().includes(city.toLowerCase()));
if (matchKey) coords = CITY_COORDS[matchKey];
}
if (coords) {
routeCoords.push(coords);
bounds.push(coords);
const isSingleTour = selectedTour !== 'ALL';
const circle = L.circleMarker(coords, {
color: '#38bdf8',
fillColor: '#facc15',
fillOpacity: 0.85,
radius: isSingleTour ? 8 : 6
});
const tourBadge = show.TOUR_NAME ? `<br><span style="color:#38bdf8; font-size:0.8rem;">🎪 ${show.TOUR_NAME}</span>` : '';
circle.bindPopup(`
<strong style="color:#facc15; font-size:0.95rem;">${isSingleTour ? '#' + (idx + 1) + ' ' : ''}${city}</strong><br>
<span style="color:#fff;">📅 ${show.DATUM || 'N/A'}</span><br>
<span style="color:#9ca3af; font-size:0.8rem;">🎭 ${show.VENUE || 'N/A'}</span>
${tourBadge}
`);
mapMarkersGroup.addLayer(circle);
}
});
if (selectedTour !== 'ALL' && selectedTour !== 'NON_TOUR' && routeCoords.length > 1) {
mapPolyline = L.polyline(routeCoords, {
color: '#38bdf8',
weight: 3,
opacity: 0.75,
dashArray: '6, 8'
}).addTo(leafletMap);
}
if (bounds.length > 0) {
leafletMap.fitBounds(bounds, { padding: [40, 40], maxZoom: 12 });
} else {
leafletMap.setView([30.0, 10.0], 2);
}
}
function renderYearBarChart(yearsMap) {
const sortedYears = Object.keys(yearsMap).sort();
const counts = sortedYears.map(y => yearsMap[y]);
new Chart(document.getElementById('chartYears'), {
type: 'bar',
data: {
labels: sortedYears,
datasets: [{
label: 'Items Count',
data: counts,
backgroundColor: '#facc15',
borderRadius: 4
}]
},
options: {
responsive: true,
maintainAspectRatio: false,
plugins: { legend: { display: false } },
scales: {
y: { grid: { color: '#1f2937' }, beginAtZero: true },
x: { grid: { display: false } }
}
}
});
}
function renderDoughnutChart(canvasId, dataMap) {
const labels = Object.keys(dataMap);
const values = Object.values(dataMap);
const colors = ['#facc15', '#38bdf8', '#f97316', '#4ade80', '#c084fc', '#ef4444', '#a855f7'];
new Chart(document.getElementById(canvasId), {
type: 'doughnut',
data: {
labels: labels,
datasets: [{
data: values,
backgroundColor: colors.slice(0, labels.length),
borderWidth: 2,
borderColor: '#111827'
}]
},
options: {
responsive: true,
maintainAspectRatio: false,
plugins: { legend: { position: 'right', labels: { boxWidth: 12, padding: 12 } } }
}
});
}
function renderBarChart(canvasId, dataMap, limit, color) {
const sorted = Object.entries(dataMap).sort((a, b) => b[1] - a[1]).slice(0, limit);
const labels = sorted.map(item => item[0]);
const values = sorted.map(item => item[1]);
new Chart(document.getElementById(canvasId), {
type: 'bar',
data: {
labels: labels,
datasets: [{ data: values, backgroundColor: color, borderRadius: 5 }]
},
options: {
responsive: true,
maintainAspectRatio: false,
indexAxis: 'y',
plugins: { legend: { display: false } },
scales: {
x: { grid: { color: '#1f2937' }, beginAtZero: true },
y: { grid: { display: false } }
}
}
});
}
