let rawTicketsData = [];
let toursGrouped = [];
let lineupsGrouped = [];
let categoriesData = [];
let editingTourIdx = -1;
let editingLineupIdx = -1;
// AUTH / SESSION LOGIC
async function checkAuthSession() {
const modal = document.getElementById('gateModal');
if (typeof supabaseClient === 'undefined' || !supabaseClient.auth) {
if (modal) modal.style.display = 'flex';
return false;
}
const { data: { session } } = await supabaseClient.auth.getSession();
if (session) {
if (modal) modal.style.display = 'none';
localStorage.setItem('jj_admin_mode', 'true');
return true;
} else {
if (modal) modal.style.display = 'flex';
localStorage.removeItem('jj_admin_mode');
return false;
}
}
async function handleSupabaseLogin() {
const email = document.getElementById('loginEmail').value.trim();
const password = document.getElementById('loginPassword').value.trim();
const errEl = document.getElementById('loginError');
if (errEl) errEl.style.display = 'none';
const { error } = await supabaseClient.auth.signInWithPassword({ email, password });
if (error) {
if (errEl) {
errEl.textContent = '❌ Invalid credentials: ' + error.message;
errEl.style.display = 'block';
}
} else {
localStorage.setItem('jj_admin_mode', 'true');
await checkAuthSession();
await loadAllAdminData();
}
}
async function lockAdminSession() {
if (typeof supabaseClient !== 'undefined' && supabaseClient.auth) {
await supabaseClient.auth.signOut();
}
localStorage.removeItem('jj_admin_mode');
window.location.href = 'index.html';
}
// TAB SWITCHING
function switchTab(tab) {
document.getElementById('tabToursBtn').classList.toggle('active', tab === 'tours');
document.getElementById('tabLineupsBtn').classList.toggle('active', tab === 'lineups');
document.getElementById('tabCategoriesBtn').classList.toggle('active', tab === 'categories');
document.getElementById('toursTab').style.display = tab === 'tours' ? 'block' : 'none';
document.getElementById('lineupsTab').style.display = tab === 'lineups' ? 'block' : 'none';
document.getElementById('categoriesTab').style.display = tab === 'categories' ? 'block' : 'none';
}
// LOAD ALL DATA
async function loadAllAdminData() {
await Promise.all([
loadTicketsData(),
loadCategoriesData()
]);
}
/* ==========================================================================
SECTION 1 & 2: TOURS & LINEUPS LOGIC
========================================================================== */
async function loadTicketsData() {
try {
const { data, error } = await supabaseClient
.from('tickets')
.select('ID_MEMORABILIA, TOUR_NAME, TOUR_ID, LINEUP');
if (error) throw error;
rawTicketsData = data || [];
processToursData();
processLineupsData();
renderToursTable();
renderLineupsTable();
} catch (err) {
console.error("Error loading tickets data:", err);
document.getElementById('toursContent').innerHTML = `<div style="color:#ef4444;">❌ Error: ${err.message}</div>`;
}
}
function processToursData() {
const map = new Map();
rawTicketsData.forEach(r => {
const name = (r.TOUR_NAME || '').trim();
if (!name) return;
const tourId = (r.TOUR_ID || '').trim();
if (!map.has(name)) {
map.set(name, { name, tourId, count: 0 });
}
const item = map.get(name);
item.count += 1;
if (!item.tourId && tourId) item.tourId = tourId;
});
toursGrouped = Array.from(map.values()).sort((a, b) => a.name.localeCompare(b.name));
}
function processLineupsData() {
const map = new Map();
rawTicketsData.forEach(r => {
const lineup = (r.LINEUP || '').trim();
if (!lineup) return;
if (!map.has(lineup)) {
map.set(lineup, { lineup, count: 0 });
}
map.get(lineup).count += 1;
});
lineupsGrouped = Array.from(map.values()).sort((a, b) => a.lineup.localeCompare(b.lineup));
}
function renderToursTable() {
const container = document.getElementById('toursContent');
if (toursGrouped.length === 0) {
container.innerHTML = '<div style="color: #888;">No tours found in database.</div>';
return;
}
let html = `
<table>
<thead>
<tr>
<th style="width: 40%;">Tour Name</th>
<th style="width: 30%;">Tour ID</th>
<th style="width: 10%;">Records</th>
<th style="text-align: right; width: 20%;">Actions</th>
</tr>
</thead>
<tbody>
`;
toursGrouped.forEach((item, idx) => {
if (editingTourIdx === idx) {
html += `
<tr style="background: rgba(245, 158, 11, 0.05);">
<td><input type="text" id="edit_tour_name_${idx}" value="${escapeHtml(item.name)}"></td>
<td><input type="text" id="edit_tour_id_${idx}" value="${escapeHtml(item.tourId)}"></td>
<td><span class="badge">${item.count} items</span></td>
<td>
<div class="edit-actions">
<button class="btn btn-green" onclick="saveTourEdit(${idx})">💾 Save</button>
<button class="btn" onclick="cancelTourEdit()">❌ Cancel</button>
</div>
</td>
</tr>
`;
} else {
html += `
<tr>
<td><strong>${escapeHtml(item.name)}</strong></td>
<td><code>${escapeHtml(item.tourId || '-')}</code></td>
<td><span class="badge">${item.count} items</span></td>
<td>
<div class="edit-actions">
<button class="btn" onclick="enableTourEdit(${idx})" title="Edit Tour">✏️</button>
</div>
</td>
</tr>
`;
}
});
html += '</tbody></table>';
container.innerHTML = html;
}
function renderLineupsTable() {
const container = document.getElementById('lineupsContent');
if (lineupsGrouped.length === 0) {
container.innerHTML = '<div style="color: #888;">No lineups found in database.</div>';
return;
}
let html = `
<table>
<thead>
<tr>
<th style="width: 70%;">Lineup Description</th>
<th style="width: 10%;">Records</th>
<th style="text-align: right; width: 20%;">Actions</th>
</tr>
</thead>
<tbody>
`;
lineupsGrouped.forEach((item, idx) => {
if (editingLineupIdx === idx) {
html += `
<tr style="background: rgba(245, 158, 11, 0.05);">
<td><textarea id="edit_lineup_${idx}">${escapeHtml(item.lineup)}</textarea></td>
<td><span class="badge">${item.count} items</span></td>
<td>
<div class="edit-actions">
<button class="btn btn-green" onclick="saveLineupEdit(${idx})">💾 Save</button>
<button class="btn" onclick="cancelLineupEdit()">❌ Cancel</button>
</div>
</td>
</tr>
`;
} else {
html += `
<tr>
<td><strong>${escapeHtml(item.lineup)}</strong></td>
<td><span class="badge">${item.count} items</span></td>
<td>
<div class="edit-actions">
<button class="btn" onclick="enableLineupEdit(${idx})" title="Edit Lineup">✏️</button>
</div>
</td>
</tr>
`;
}
});
html += '</tbody></table>';
container.innerHTML = html;
}
function enableTourEdit(idx) { editingTourIdx = idx; renderToursTable(); }
function cancelTourEdit() { editingTourIdx = -1; renderToursTable(); }
async function saveTourEdit(idx) {
const oldItem = toursGrouped[idx];
const newName = document.getElementById(`edit_tour_name_${idx}`).value.trim();
const newTourId = document.getElementById(`edit_tour_id_${idx}`).value.trim();
if (!newName) { alert("Tour name cannot be empty!"); return; }
if (!confirm(`Are you sure you want to update all ${oldItem.count} records with Tour "${oldItem.name}" to "${newName}"?`)) return;
try {
const { error } = await supabaseClient.from('tickets').update({ TOUR_NAME: newName, TOUR_ID: newTourId || null }).eq('TOUR_NAME', oldItem.name);
if (error) throw error;
editingTourIdx = -1;
await loadTicketsData();
} catch (err) { alert('❌ Error updating tour: ' + err.message); }
}
function enableLineupEdit(idx) { editingLineupIdx = idx; renderLineupsTable(); }
function cancelLineupEdit() { editingLineupIdx = -1; renderLineupsTable(); }
async function saveLineupEdit(idx) {
const oldItem = lineupsGrouped[idx];
const newLineup = document.getElementById(`edit_lineup_${idx}`).value.trim();
if (!newLineup) { alert("Lineup cannot be empty!"); return; }
if (newLineup === oldItem.lineup) { cancelLineupEdit(); return; }
if (!confirm(`Are you sure you want to update this lineup across ALL ${oldItem.count} records in database?`)) return;
try {
const { error } = await supabaseClient.from('tickets').update({ LINEUP: newLineup }).eq('LINEUP', oldItem.lineup);
if (error) throw error;
editingLineupIdx = -1;
await loadTicketsData();
} catch (err) { alert('❌ Error updating lineup: ' + err.message); }
}

SECTION 3: CATEGORIES & SUBCATEGORIES LOGIC (WITH PUBLIC VISIBILITY)
========================================================================== */
async function loadCategoriesData() {
const container = document.getElementById('categoriesContent');
try {
const { data, error } = await supabaseClient
.from('categories')
.select('*')
.order('display_order', { ascending: true, nullsFirst: false })
.order('id', { ascending: true });
if (error) throw error;
categoriesData = data || [];
renderCategoriesList();
} catch (err) {
container.innerHTML = `<div style="color:#ef4444; padding: 12px; background: rgba(239, 68, 68, 0.1); border-radius: 6px;">❌ Error loading categories: ${err.message}</div>`;
}
}
function renderCategoriesList() {
const container = document.getElementById('categoriesContent');
if (categoriesData.length === 0) {
container.innerHTML = '<div class="card" style="color: #888;">No categories found in database.</div>';
return;
}
container.innerHTML = categoriesData.map((cat, idx) => {
const subList = Array.isArray(cat.subcategories) ? cat.subcategories : [];
const isFirst = idx === 0;
const isLast = idx === categoriesData.length - 1;
const isPublic = cat.is_public !== false;
return `
<div class="card">
<div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 12px; border-bottom: 1px solid var(--border-color); padding-bottom: 8px;">
<div style="display: flex; align-items: center; gap: 8px;">
<button class="btn" onclick="moveCategory(${idx}, -1)" ${isFirst ? 'disabled style="opacity:0.3;cursor:default;"' : ''} title="Move Up">⬆️</button>
<button class="btn" onclick="moveCategory(${idx}, 1)" ${isLast ? 'disabled style="opacity:0.3;cursor:default;"' : ''} title="Move Down">⬇️</button>
<strong style="font-size: 1.1rem; color: #fff; margin-left: 6px;">${escapeHtml(cat.name)}</strong>
</div>
<div style="display: flex; gap: 8px; align-items: center;">
<button class="visibility-btn ${isPublic ? 'public' : 'hidden'}" onclick="toggleCategoryVisibility('${cat.id}', ${isPublic})" title="Click to toggle public status">
${isPublic ? '👁️ Public' : '🙈 Hidden'}
</button>
<button class="btn" onclick="renameCategory('${cat.id}', '${escapeJsStr(cat.name)}')">✏️ Rename</button>
<button class="btn btn-danger" onclick="deleteCategory('${cat.id}', '${escapeJsStr(cat.name)}')">🗑️ Delete</button>
</div>
</div>
<label style="font-size: 0.75rem; color: #888; font-weight: 600; text-transform: uppercase;">SUBCATEGORIES:</label>
<div class="subcat-list">
${subList.length > 0 ? subList.map((sub, sIdx) => {
const subVal = typeof sub === 'object' ? (sub.value || sub.name) : sub;
const isSubFirst = sIdx === 0;
const isSubLast = sIdx === subList.length - 1;
return `
<div class="subcat-chip">
<span class="arrow" onclick="moveSubcategory('${cat.id}', ${sIdx}, -1)" style="${isSubFirst ? 'visibility:hidden;' : ''}" title="Move left">◀</span>
<span>${escapeHtml(subVal)}</span>
<span class="arrow" onclick="moveSubcategory('${cat.id}', ${sIdx}, 1)" style="${isSubLast ? 'visibility:hidden;' : ''}" title="Move right">▶</span>
<span class="delete-btn" onclick="removeSubcategory('${cat.id}', '${escapeJsStr(subVal)}')" title="Remove subcategory">×</span>
</div>
`;
}).join('') : '<span style="font-size: 0.8rem; color: #555; font-style: italic;">No subcategories</span>'}
</div>
<div style="display: flex; gap: 8px; margin-top: 14px;">
<input type="text" id="sub_input_${cat.id}" placeholder="New subcategory name..." style="margin-bottom: 0;" onkeydown="if(event.key==='Enter') addSubcategory('${cat.id}')">
<button class="btn" onclick="addSubcategory('${cat.id}')" style="white-space: nowrap;">+ Add Subcategory</button>
</div>
</div>
`;
}).join('');
}
async function toggleCategoryVisibility(catId, currentStatus) {
const newStatus = !currentStatus;
const { error } = await supabaseClient.from('categories').update({ is_public: newStatus }).eq('id', catId);
if (error) {
alert('Error changing category visibility: ' + error.message);
} else {
await loadCategoriesData();
}
}
async function moveCategory(index, direction) {
const targetIndex = index + direction;
if (targetIndex < 0 || targetIndex >= categoriesData.length) return;
const catA = categoriesData[index];
const catB = categoriesData[targetIndex];
const orderA = catA.display_order ?? index;
const orderB = catB.display_order ?? targetIndex;
const newOrderA = orderA === orderB ? targetIndex : orderB;
const newOrderB = orderA === orderB ? index : orderA;
try {
await Promise.all([
supabaseClient.from('categories').update({ display_order: newOrderA }).eq('id', catA.id),
supabaseClient.from('categories').update({ display_order: newOrderB }).eq('id', catB.id)
]);
await loadCategoriesData();
} catch (err) { alert('Error updating category order: ' + err.message); }
}
async function moveSubcategory(catId, subIndex, direction) {
const cat = categoriesData.find(c => String(c.id) === String(catId));
if (!cat) return;
const subs = Array.isArray(cat.subcategories) ? [...cat.subcategories] : [];
const targetIndex = subIndex + direction;
if (targetIndex < 0 || targetIndex >= subs.length) return;
const temp = subs[subIndex];
subs[subIndex] = subs[targetIndex];
subs[targetIndex] = temp;
const { error } = await supabaseClient.from('categories').update({ subcategories: subs }).eq('id', catId);
if (error) alert('Error updating subcategory order: ' + error.message);
else loadCategoriesData();
}
async function addCategory() {
const input = document.getElementById('newCategoryInput');
const name = input.value.trim();
if (!name) return;
const nextOrder = categoriesData.length > 0 ? (Math.max(...categoriesData.map(c => c.display_order || 0)) + 1) : 1;
const { error } = await supabaseClient.from('categories').insert([{ name, subcategories: [], display_order: nextOrder, is_public: true }]);
if (error) alert('Error adding category: ' + error.message);
else { input.value = ''; loadCategoriesData(); }
}
async function renameCategory(id, currentName) {
const newName = prompt('New name for category:', currentName);
if (!newName || newName.trim() === '' || newName === currentName) return;
const { error } = await supabaseClient.from('categories').update({ name: newName.trim() }).eq('id', id);
if (error) alert('Error renaming category: ' + error.message);
else loadCategoriesData();
}
async function deleteCategory(id, name) {
if (!confirm(`Are you sure you want to delete category "${name}"?`)) return;
const { error } = await supabaseClient.from('categories').delete().eq('id', id);
if (error) alert('Error deleting category: ' + error.message);
else loadCategoriesData();
}
async function addSubcategory(catId) {
const input = document.getElementById(`sub_input_${catId}`);
const val = input.value.trim();
if (!val) return;
const cat = categoriesData.find(c => String(c.id) === String(catId));
if (!cat) return;
const currentSubs = Array.isArray(cat.subcategories) ? cat.subcategories.map(s => typeof s === 'object' ? s.value : s) : [];
if (currentSubs.includes(val)) { alert('Subcategory already exists!'); return; }
const updatedSubs = [...currentSubs, val];
const { error } = await supabaseClient.from('categories').update({ subcategories: updatedSubs }).eq('id', catId);
if (error) alert('Error adding subcategory: ' + error.message);
else { input.value = ''; loadCategoriesData(); }
}
async function removeSubcategory(catId, subName) {
const cat = categoriesData.find(c => String(c.id) === String(catId));
if (!cat) return;
const currentSubs = Array.isArray(cat.subcategories) ? cat.subcategories.map(s => typeof s === 'object' ? s.value : s) : [];
const updatedSubs = currentSubs.filter(s => String(s) !== String(subName));
const { error } = await supabaseClient.from('categories').update({ subcategories: updatedSubs }).eq('id', catId);
if (error) alert('Error removing subcategory: ' + error.message);
else loadCategoriesData();
}
// UTILS
function escapeHtml(str) {
return String(str || '').replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}
function escapeJsStr(str) {
return String(str || '').replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/"/g, '\\"');
}
